J = 1
def Hamiltonian(spins, rows, J):
    E = 0
    for i in range(rows):
        for j in range(cols):
            S = spins[i, j]
            neighbors = (spins[(i+1) % 10, j] + spins[i, (j+1) % 10] + spins[(i-1) % 10, j] + spins[i, (j-1) % 10])
            E += -J * S * neighbors
    return E/2
