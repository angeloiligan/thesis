import numpy as np
import matplotlib.pyplot as plt

# down = -1
# up = 1

rows = 10
cols = 10

spins = np.random.choice([-1,1], size=(rows,cols))

J = 1
T = 2.0
kB = 1.0

def Hamiltonian(spins, rows, cols, J):
    E = 0
    for i in range(rows):
        for j in range(cols):
            S = spins[i,j]
            neighbors = (spins[(i+1)%rows, j] + spins[(i-1)%rows, j] + spins[i,(j+1)%cols] + spins[i,(j-1)%cols])
            E += -J * S * neighbors

    return E/2

steps = 10000

energies = []
magnetizations = []

for step in range(steps):
    
    Old_energy = Hamiltonian(spins, rows, cols, J)

    i = np.random.randint(rows)
    j = np.random.randint(cols)

    spins[i,j] *= -1

    New_energy = Hamiltonian(spins, rows, cols, J)

    OE = Old_energy
    NE = New_energy

    Energy_difference = NE - OE

    if Energy_difference <=0:
        accept = True
    else:
        probability = np.exp(-Energy_difference/(kB*T))
        r = np.random.rand()
        accept = r < probability

    if not accept:
        spins[i,j] *= -1

    magnetization = np.sum(spins)/(rows*cols)
    energies.append(Hamiltonian(spins, rows, cols, J))
    magnetizations.append(magnetization)
